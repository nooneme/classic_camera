移动端 RAW 级 AI 滤镜学习与应用 完整技术方案
一、 架构概览与数据流
整套系统分为四个核心模块，数据流如下：

样本采集与学习 (C++/NDK)：用户传入对齐的“原图/滤镜图”对。C++ 算法提取像素对，通过散点三线性插值将颜色映射关系分配到 33×33×33 的 3D 网格中。同时利用体素命中法计算色彩覆盖率。
LUT 数据打包：将 C++ 生成的 float[] 数组转换为一张 1089×33 像素的 2D Bitmap，供 OpenGL 采样。
RAW 管线注入 (OpenGL ES 3.0)：在您现有的 RAW Bayer -> Demosaic -> CCM -> Gamma -> S-curve 管线最末端，注入 LUT 采样逻辑，实现零延迟预览和高质量出图。
3D 色彩空间可视化：将学习到的色彩覆盖率数据转换为 3D 点云，用 OpenGL 渲染为一个旋转的发光立方体，给用户提供直观的 AI 学习反馈。
二、 核心算法层：C++ 散点拟合与覆盖率检测
目标：不依赖深度学习框架，在 100ms 内完成 LUT 生成，并解决样本分布不均导致的 LUT 空洞和色块断层问题。

策略说明
体素命中记录：计算覆盖率时，不依赖插值权重，只要像素落入网格区间即标记命中。
锚点注入：在 RGB 立方体 8 个顶点注入高权重虚拟样本，防止极端颜色崩溃。
L2 正则化收缩：样本稀疏的网格点，其映射颜色向原图颜色收缩，防止噪点放大。
C++ 代码实现 (通过 JNI 暴露给 Kotlin)

// lut_generator.cpp
#include <jni.h>
#include <vector>
#include <cmath>
#include <algorithm>

const int LUT_SIZE = 33;
const int TOTAL_NODES = LUT_SIZE * LUT_SIZE * LUT_SIZE;

struct Vec3 { float r, g, b; };

extern "C" JNIEXPORT jfloat JNICALL
Java_com_example_yourapp_LutEngine_generateLutAndCheckCoverage(
    JNIEnv* env, jobject thiz,
    jintArray orig_pixels, jintArray filt_pixels,
    jint num_pixels, jfloatArray out_lut_array) {

    std::vector<Vec3> colorSum(TOTAL_NODES, {0,0,0});
    std::vector<float> weightSum(TOTAL_NODES, 0.0f);
    std::vector<bool> covered(TOTAL_NODES, false);

    // 1. 注入色彩锚点 (防止边缘崩溃)
    auto addAnchor = [&](int r, int g, int b, float weight) {
        float idx_r = r / 255.0f * (LUT_SIZE - 1);
        float idx_g = g / 255.0f * (LUT_SIZE - 1);
        float idx_b = b / 255.0f * (LUT_SIZE - 1);
        int idx = static_cast<int>(idx_r) + static_cast<int>(idx_g) * LUT_SIZE + static_cast<int>(idx_b) * LUT_SIZE * LUT_SIZE;
        colorSum[idx] += {(float)r/255.0f, (float)g/255.0f, (float)b/255.0f};
        weightSum[idx] += weight;
    };
    addAnchor(0, 0, 0, 1000.0f);       addAnchor(255, 255, 255, 1000.0f);
    addAnchor(255, 0, 0, 500.0f);      addAnchor(0, 255, 0, 500.0f);
    addAnchor(0, 0, 255, 500.0f);      addAnchor(255, 255, 0, 500.0f);
    addAnchor(0, 255, 255, 500.0f);    addAnchor(255, 0, 255, 500.0f);

    // 2. 读取像素并散点收集
    jint* orig = env->GetIntArrayElements(orig_pixels, nullptr);
    jint* filt = env->GetIntArrayElements(filt_pixels, nullptr);

    for (int i = 0; i < num_pixels; ++i) {
        int oR = (orig[i] >> 16) & 0xFF; int oG = (orig[i] >> 8) & 0xFF; int oB = orig[i] & 0xFF;
        int fR = (filt[i] >> 16) & 0xFF; int fG = (filt[i] >> 8) & 0xFF; int fB = filt[i] & 0xFF;

        // 记录体素命中 (用于覆盖率计算)
        int vx = std::min(32, (oR * LUT_SIZE) / 256);
        int vy = std::min(32, (oG * LUT_SIZE) / 256);
        int vz = std::min(32, (oB * LUT_SIZE) / 256);
        covered[vx + vy * LUT_SIZE + vz * LUT_SIZE * LUT_SIZE] = true;

        // 三线性插值分配
        float fx = oR / 255.0f * (LUT_SIZE - 1);
        float fy = oG / 255.0f * (LUT_SIZE - 1);
        float fz = oB / 255.0f * (LUT_SIZE - 1);
        int x0 = floor(fx), y0 = floor(fy), z0 = floor(fz);
        int x1 = std::min(x0 + 1, LUT_SIZE - 1);
        int y1 = std::min(y0 + 1, LUT_SIZE - 1);
        int z1 = std::min(z0 + 1, LUT_SIZE - 1);

        float dx = fx - x0, dy = fy - y0, dz = fz - z0;
        float weights[8] = {
            (1-dx)*(1-dy)*(1-dz), dx*(1-dy)*(1-dz), (1-dx)*dy*(1-dz), dx*dy*(1-dz),
            (1-dx)*(1-dy)*dz, dx*(1-dy)*dz, (1-dx)*dy*dz, dx*dy*dz
        };
        int indices[8] = {
            x0 + y0*LUT_SIZE + z0*LUT_SIZE*LUT_SIZE, x1 + y0*LUT_SIZE + z0*LUT_SIZE*LUT_SIZE,
            x0 + y1*LUT_SIZE + z0*LUT_SIZE*LUT_SIZE, x1 + y1*LUT_SIZE + z0*LUT_SIZE*LUT_SIZE,
            x0 + y0*LUT_SIZE + z1*LUT_SIZE*LUT_SIZE, x1 + y0*LUT_SIZE + z1*LUT_SIZE*LUT_SIZE,
            x0 + y1*LUT_SIZE + z1*LUT_SIZE*LUT_SIZE, x1 + y1*LUT_SIZE + z1*LUT_SIZE*LUT_SIZE
        };

        for(int j=0; j<8; ++j) {
            colorSum[indices[j]].r += weights[j] * (fR / 255.0f);
            colorSum[indices[j]].g += weights[j] * (fG / 255.0f);
            colorSum[indices[j]].b += weights[j] * (fB / 255.0f);
            weightSum[indices[j]] += weights[j];
        }
    }
    env->ReleaseIntArrayElements(orig_pixels, orig, 0);
    env->ReleaseIntArrayElements(filt_pixels, filt, 0);

    // 3. 计算最终颜色与覆盖率
    std::vector<Vec3> finalLut(TOTAL_NODES);
    int coveredCount = 0;
    jfloat lutOut[TOTAL_NODES * 3];

    for (int i = 0; i < TOTAL_NODES; ++i) {
        if (covered[i]) coveredCount++;
        
        int z = i / (LUT_SIZE * LUT_SIZE);
        int y = (i % (LUT_SIZE * LUT_SIZE)) / LUT_SIZE;
        int x = i % LUT_SIZE;
        Vec3 identity = {x / (float)(LUT_SIZE - 1), y / (float)(LUT_SIZE - 1), z / (float)(LUT_SIZE - 1)};

        if (weightSum[i] > 0.001f) {
            float confidence = std::min(1.0f, weightSum[i] / 5.0f); // 置信度饱和
            Vec3 sampled = {colorSum[i].r/weightSum[i], colorSum[i].g/weightSum[i], colorSum[i].b/weightSum[i]};
            finalLut[i].r = confidence * sampled.r + (1 - confidence) * identity.r;
            finalLut[i].g = confidence * sampled.g + (1 - confidence) * identity.g;
            finalLut[i].b = confidence * sampled.b + (1 - confidence) * identity.b;
        } else {
            finalLut[i] = identity; // 无样本空洞点保持原色
        }
        
        lutOut[i * 3] = finalLut[i].r;
        lutOut[i * 3 + 1] = finalLut[i].g;
        lutOut[i * 3 + 2] = finalLut[i].b;
    }

    // 拷贝到 Java 数组
    env->SetFloatArrayRegion(out_lut_array, 0, TOTAL_NODES * 3, lutOut);
    
    return (float)coveredCount / TOTAL_NODES; // 返回 0.0 - 1.0 覆盖率
}
三、 数据转换层：LUT 数组转 2D 纹理 Bitmap
Kotlin 层接收到 C++ 返回的 lutFloatArray 后，将其转换为 OpenGL 可识别的 1089×33 ARGB_8888 Bitmap。


// LutUtils.kt
import android.graphics.Bitmap
import android.graphics.Color

object LutUtils {
    const val LUT_SIZE = 33

    fun createLutBitmap(lutFloatArray: FloatArray): Bitmap {
        val width = LUT_SIZE * LUT_SIZE // 1089
        val height = LUT_SIZE          // 33
        val lutBitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)

        val pixels = IntArray(width * height)

        for (y in 0 until height) {
            for (x in 0 until width) {
                val bSlice = x / LUT_SIZE
                val rIndex = x % LUT_SIZE
                val gIndex = y

                // 计算在 1D 数组中的索引
                val index = (bSlice * LUT_SIZE * LUT_SIZE + gIndex * LUT_SIZE + rIndex) * 3

                val r = (lutFloatArray[index] * 255f).toInt().coerceIn(0, 255)
                val g = (lutFloatArray[index + 1] * 255f).toInt().coerceIn(0, 255)
                val b = (lutFloatArray[index + 2] * 255f).toInt().coerceIn(0, 255)

                pixels[y * width + x] = Color.argb(255, r, g, b)
            }
        }
        lutBitmap.setPixels(pixels, 0, width, 0, 0, width, height)
        return lutBitmap
    }
}
四、 RAW 渲染管线注入：OpenGL Shader 整合
整合原则：不改动您现有的 fix, cfaCh, Demosaic, CCM 逻辑。在所有最终输出 Program（progBilinear, progHA, progMHC）的 S-curve 之后追加 LUT 采样。

1. GLSL Shader 代码修改
在您的 Fragment Shader 字符串中追加以下逻辑：


#version 300 es
precision highp float;

// ... (您原有的 uniform 和 varying 声明) ...

// === 新增：LUT 相关 Uniform ===
uniform sampler2D uLutTexture;
uniform float uLutSize;       // 33.0
uniform bool uEnableLut;      // 开关

// === 新增：LUT 查找函数 ===
vec3 applyLUT(vec3 color) {
    vec3 lutCoord = color * (uLutSize - 1.0);
    float blueSlice = floor(lutCoord.b);
    float blueOffset = lutCoord.b - blueSlice;
    
    // 2D 纹理坐标计算 (含 0.5 像素偏移以精确采样纹素中心)
    float rCoord1 = (lutCoord.r + blueSlice * uLutSize + 0.5) / (uLutSize * uLutSize);
    float rCoord2 = (lutCoord.r + min(blueSlice + 1.0, uLutSize - 1.0) * uLutSize + 0.5) / (uLutSize * uLutSize);
    float gCoord = (lutCoord.g + 0.5) / uLutSize;
    
    vec3 lutColor1 = texture(uLutTexture, vec2(rCoord1, gCoord)).rgb;
    vec3 lutColor2 = texture(uLutTexture, vec2(rCoord2, gCoord)).rgb;
    
    return mix(lutColor1, lutColor2, blueOffset);
}

void main() {
    // ... (您原有的 RAW 解码逻辑: fix, cfaCh, Demosaic) ...
    vec3 rgb = uCCM * rgb_demosaic;
    rgb = clamp(rgb, 0.0, 1.0);
    rgb = pow(rgb, vec3(1.0/2.2));
    rgb = mix(rgb, smoothstep(0.0, 1.0, rgb), 0.20); // 您的 S-curve
    
    // === 核心插入点 ===
    if (uEnableLut) {
        rgb = applyLUT(rgb);
    }
    
    fragColor = vec4(rgb, 1.0);
}

五、 进阶 UI：3D 色彩空间覆盖率可视化
将 C++ 层返回的 covered 布尔数组（或重构为 0/1 的 byte[]）传回 Java 层，生成点云坐标，用 OpenGL 绘制 3D 发光立方体。

1. 点云数据生成 (Kotlin)

// 3DVisualizer.kt
fun generateCoveragePointCloud(coveredArray: BooleanArray, lutSize: Int = 33): FloatArray {
    val points = mutableListOf<Float>()
    for (z in 0 until lutSize) {
        for (y in 0 until lutSize) {
            for (x in 0 until lutSize) {
                if (coveredArray[x + y * lutSize + z * lutSize * lutSize]) {
                    // 归一化到 -0.5 到 0.5 的 3D 空间
                    points.add((x / (lutSize - 1).toFloat()) - 0.5f) // X
                    points.add((y / (lutSize - 1).toFloat()) - 0.5f) // Y
                    points.add((z / (lutSize - 1).toFloat()) - 0.5f) // Z
                }
            }
        }
    }
    return points.toFloatArray()
}
2. OpenGL 渲染点云 (独立于 RAW 管线的简易 Renderer)
你可以将这个点云渲染在一个透明的 GLSurfaceView 叠加在 UI 上。

Vertex Shader (点云)


#version 300 es
in vec3 aPosition;
uniform mat4 uMVPMatrix; // 旋转矩阵
out vec3 vColor;

void main() {
    gl_Position = uMVPMatrix * vec4(aPosition, 1.0);
    // 根据位置生成颜色 (RGB 映射到 XYZ)
    vColor = aPosition + 0.5; 
    gl_PointSize = 3.0; // 点大小
}
Fragment Shader (发光点)


#version 300 es
precision mediump float;
in vec3 vColor;
out vec4 fragColor;

void main() {
    // 画圆形的点并带发光效果
    vec2 coord = gl_PointCoord - vec2(0.5);
    float dist = length(coord);
    if(dist > 0.5) discard;
    
    float alpha = smoothstep(0.5, 0.0, dist);
    fragColor = vec4(vColor, alpha);
}
